Method:
[server only]
boolean TryEnterPartyBossMap(table userList,string bossName,integer difficult)
{
	-- 원본 맵의 이름을 지정합니다. 이 예제는 보스 이름 + Base+ 난이도(예: ZakumBase1)로 가정합니다.
	local bossMapName = bossName .. "Base" .. tostring(math.tointeger(difficult))
	local newMapName = bossName .. tostring(_UtilLogic:RandomInteger())
	
	-- 동적 맵은 중복된 이름을 사용할 수 없습니다. 이름이 중복되는 경우 새로운 이름을 만듭니다.
	while not self:IsValidDynamicMapName(newMapName) do
		newMapName = bossName .. tostring(_UtilLogic:RandomInteger())
	end
	
	-- 동적 맵을 생성합니다.
	local resultCode = _DynamicMapService:CreateDynamicMap(bossMapName, newMapName)
	
	if resultCode ~= DynamicMapResultCode.Success then
		-- 동적 맵 생성에 실패한 경우 에러 코드를 알려줍니다.
		self:OnError(resultCode)
		return false
	end
	
	local reservedUserEntities = {}
	local reserved = true
	
	-- 새로운 맵으로 이동할 수 있도록 예약합니다.
	for index, userId in pairs(userList) do
		local userEntity = _UserService:GetUserEntityByUserId(userId)
		reserved = reserved & _TeleportService:ReserveTeleportToMapPosition(userEntity, Vector3.zero, newMapName)
		if not reserved then
			break
		end
		table.insert(reservedUserEntities, userEntity)
	end
	
	-- 맵 이동 예약을 실패 시의 처리 방식입니다.
	if not reserved then
		for index, userEntity in pairs(reservedUserEntities) do
			_TeleportService:UnReserveTeleport(userEntity)
		end
		self:DestroyDynamicMap(newMapName)
		return false
	end
	
	-- 동적 맵 관리에 필요한 정보를 생성한 후 예약된 맵 이동을 실행합니다.
	local userIdSet = {}
	for index, userId in pairs(userList) do
		userIdSet[userId] = true
	end
	
	self.userIdSetByDynamicMapName[newMapName] = userIdSet
	self.dynamicMapNameSet[newMapName] = true
	_TeleportService:TeleportReservedEntities()
	
	return true
}

[server only]
void OnBossCleared(string mapName)
{
	if self.dynamicMapNameSet[mapName] == nil then
		return
	end
	
	-- 동적 맵에 존재하는 모든 유저를 'Lobby' 맵으로 이동시킵니다.
	local userEntityList = _UserService:GetUsersByMapName(mapName)
	for i, userEntity in pairs(userEntityList) do
		_TeleportService:ReserveTeleportToMapPosition(userEntity, Vector3.zero, "Lobby")
	end
	_TeleportService:TeleportReservedEntities()
	
	self:DestroyDynamicMap(mapName)
}

[server only]
void DestroyDynamicMap(string mapName)
{
	-- 동적 맵을 파괴합니다.
	local resultCode = _DynamicMapService:DestroyDynamicMap(mapName)
	
	if resultCode ~= DynamicMapResultCode.Success then
		self:OnError(resultCode)
		return
	end
	
	-- 파괴한 동적 맵과 관련된 정보를 제거합니다.
	self.dynamicMapNameSet[mapName] = nil
	self.userIdSetByDynamicMapName[mapName] = nil
}

Event Handler:
[server only] [service: UserService]
HandleUserLeaveEven (UserLeaveEvent event)
{
	-- 동적 맵에 존재하던 유저가 월드에서 퇴장 시 처리 방식입니다.
	local UserId = event.UserId
	local userEntity = _UserService:GetUserEntityByUserId(UserId)
	local mapName = userEntity.CurrentMapName
	
	-- 유저가 있던 맵이 동적 맵인지 확인합니다.
	if self.dynamicMapNameSet[mapName] == nil then
		return
	end
	
	-- 동적 맵에 입장해 있는 유저 목록에서 해당 유저를 제거합니다.
	local userSet = self.userIdSetByDynamicMapName[mapName]
	userSet[UserId] = nil
	
	-- 동적 맵에 남은 유저가 없으면 동적 맵을 파괴합니다.
	local isEmpty = (next(userSet, nil) == nil)
	if isEmpty then
		self:DestroyDynamicMap(mapName)
	end
}
