Property:
[None]
integer GameIdx = "0"

Method:
[server only]
void EnterGame (table userIds)
{
	-- 중복되지 않는 인스턴스 룸 식별자를 만듭니다.
	local roomKey = "Game" .. tostring(self.gameIdx)
	self.gameIdx = self.GameIdx + 1
	 
	-- 맵 GameMap01, GameMap02를 사용하는 인스턴스 룸을 생성합니다.
	-- 이 맵들은 InstanceMap 속성이 true로 지정되어야 합니다.
	-- 정적 룸에서 실행되어야 합니다.
	local instanceRoom = _RoomService:CreateInstanceRoom(roomKey, {"GameMap01", "GameMap02"})
	 
	-- 이미 사용 중인 식별자이거나 자주 요청했을 경우 생성에 실패할 수 있습니다.
	if instanceRoom == nil then
		log_error("Cannot create instance room. roomKey: " .. roomKey)
		return
	end
	 
	-- 생성된 인스턴스 룸의 특정 맵으로 유저들을 입장시킵니다.
	-- _RoomService:MoveUsersToInstanceRoom() 함수를 사용할 수도 있습니다.
	instanceRoom:MoveUsers(userIds, "GameMap01")
}
Method:
[server only]
void ReturnToLobby (table userIds)
{
        -- 정적 룸의 특정 맵으로 유저들을 입장시킵니다.
        _RoomService:MoveUsersToStaticRoom(userIds, "LobbyMap")
}
